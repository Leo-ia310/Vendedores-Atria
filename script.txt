/** Crea las hojas y encabezados que falten. Idempotente. */
function asegurarHojas() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  Object.keys(SHEETS).forEach(function (nombre) {
    var hoja = ss.getSheetByName(nombre);
    if (!hoja) hoja = ss.insertSheet(nombre);
    var headers = SHEETS[nombre];
    var actuales = hoja.getRange(1, 1, 1, Math.max(headers.length, hoja.getLastColumn() || 1))
      .getValues()[0];
    var faltan = headers.some(function (h, i) { return actuales[i] !== h; });
    if (faltan) {
      hoja.getRange(1, 1, 1, headers.length).setValues([headers]);
      hoja.getRange(1, 1, 1, headers.length).setFontWeight('bold')
        .setBackground('#2b1f3a').setFontColor('#ffffff');
      hoja.setFrozenRows(1);
    }
  });
  // Guarda el ID de la hoja como propiedad para el backend.
  PropertiesService.getScriptProperties()
    .setProperty('SPREADSHEET_ID', ss.getId());
}

function asegurarConfigDefaults() {
  Object.keys(CONFIG_DEFAULTS).forEach(function (clave) {
    if (configGet(clave) === null) {
      configSet(clave, CONFIG_DEFAULTS[clave][0], CONFIG_DEFAULTS[clave][1]);
    }
  });
}

function menuConfigurarSecretos() {
  var ui = SpreadsheetApp.getUi();
  var props = PropertiesService.getScriptProperties();

  if (!props.getProperty('PEPPER')) {
    props.setProperty('PEPPER', generarToken(32));
  }

  var email = ui.prompt('Admin inicial', 'Correo del administrador:', ui.ButtonSet.OK_CANCEL);
  if (email.getSelectedButton() !== ui.Button.OK) return;
  var pass = ui.prompt('Admin inicial', 'Contraseña del administrador:', ui.ButtonSet.OK_CANCEL);
  if (pass.getSelectedButton() !== ui.Button.OK) return;

  crearUsuario(email.getResponseText().trim(), pass.getResponseText(), 'admin', '', false);
  ui.alert('Administrador creado. PEPPER generado y guardado.');
}

/* ============================================================================================
 * 3. ENTRADA HTTP: doGet / doPost + ROUTER
 * ==========================================================================================*/

function doGet(e) {
  // Health-check y verificación pública de certificados.
  var accion = (e && e.parameter && e.parameter.action) || 'ping';
  try {
    if (accion === 'verificarCertificado') {
      return respuesta(true, verificarCertificado(e.parameter.codigo || ''));
    }
    return respuesta(true, { servicio: 'Academia Comercial ARCA', estado: 'ok', version: '1.0' });
  } catch (err) {
    return respuesta(false, null, 'ERROR', err.message);
  }
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(20000);
  } catch (errLock) {
    return respuesta(false, null, 'BUSY', 'Servidor ocupado, reintenta.');
  }

  try {
    var cuerpo = {};
    if (e && e.postData && e.postData.contents) {
      cuerpo = JSON.parse(e.postData.contents);
    }
    var accion = cuerpo.action;
    var token = cuerpo.token || null;
    var payload = cuerpo.payload || {};
    var contexto = {
      ip: (e && e.parameter && e.parameter.ip) || '',
      userAgent: payload.__ua || ''
    };
    return router(accion, payload, token, contexto);
  } catch (err) {
    registrarError(err);
    return respuesta(false, null, 'ERROR', 'Error del servidor: ' + err.message);
  } finally {
    lock.releaseLock();
  }
}

/** Enruta cada acción. Acciones con prefijo "admin" exigen rol admin. */
function router(accion, payload, token, ctx) {
  switch (accion) {
    // --- Público ---
    case 'ping': return respuesta(true, { estado: 'ok' });
    case 'configPublica': return respuesta(true, configPublica());
    case 'registrarCandidato': return accRegistrarCandidato(payload, ctx);
    case 'login': return accLogin(payload, ctx);
    case 'solicitarRecuperacion': return accSolicitarRecuperacion(payload, ctx);
    case 'restablecerPassword': return accRestablecerPassword(payload, ctx);
    case 'verificarCertificado': return respuesta(true, verificarCertificado(payload.codigo));

    // --- Requieren sesión ---
    case 'logout': return accLogout(token);
    case 'sesionActual': return accSesionActual(token);
    case 'cambiarPassword': return accCambiarPassword(payload, token);

    // Academia (candidato)
    case 'obtenerModulos': return conSesion(token, function (u) { return accObtenerModulos(u); });
    case 'obtenerProgreso': return conSesion(token, function (u) { return accObtenerProgreso(u); });
    case 'guardarProgreso': return conSesion(token, function (u) { return accGuardarProgreso(u, payload); });
    case 'obtenerExamen': return conSesion(token, function (u) { return accObtenerExamen(u, payload); });
    case 'enviarExamen': return conSesion(token, function (u) { return accEnviarExamen(u, payload); });
    case 'enviarSimulacion': return conSesion(token, function (u) { return accEnviarSimulacion(u, payload); });
    case 'aceptarTerminos': return conSesion(token, function (u) { return accAceptarTerminos(u, payload, ctx); });
    case 'estadoCertificacion': return conSesion(token, function (u) { return accEstadoCertificacion(u); });
    case 'certificar': return conSesion(token, function (u) { return accCertificar(u, ctx); });

    // Vendedor
    case 'dashboardVendedor': return conRol(token, 'vendedor', function (u) { return accDashboardVendedor(u); });
    case 'listarProspectos': return conRol(token, 'vendedor', function (u) { return accListarProspectos(u); });
    case 'crearProspecto': return conRol(token, 'vendedor', function (u) { return accCrearProspecto(u, payload); });
    case 'actualizarProspecto': return conRol(token, 'vendedor', function (u) { return accActualizarProspecto(u, payload); });
    case 'registrarActividad': return conRol(token, 'vendedor', function (u) { return accRegistrarActividad(u, payload); });
    case 'listarVentas': return conRol(token, 'vendedor', function (u) { return accListarVentas(u); });
    case 'registrarVenta': return conRol(token, 'vendedor', function (u) { return accRegistrarVenta(u, payload); });
    case 'listarComisiones': return conRol(token, 'vendedor', function (u) { return accListarComisiones(u); });
    case 'actualizarPerfil': return conSesion(token, function (u) { return accActualizarPerfil(u, payload); });

    // Chatbot
    case 'registrarChat': return accRegistrarChat(payload, token);
    case 'registrarPreguntaNoResuelta': return accRegistrarPreguntaNoResuelta(payload, token);

    // --- Admin ---
    case 'adminListar': return conRol(token, 'admin', function () { return accAdminListar(payload); });
    case 'adminCandidato': return conRol(token, 'admin', function (u) { return accAdminCandidato(u, payload); });
    case 'adminActualizarVenta': return conRol(token, 'admin', function (u) { return accAdminActualizarVenta(u, payload); });
    case 'adminCalcularComision': return conRol(token, 'admin', function (u) { return accAdminCalcularComision(u, payload); });
    case 'adminMarcarComisionPagada': return conRol(token, 'admin', function (u) { return accAdminMarcarComisionPagada(u, payload); });
    case 'adminRegistrarRenovacion': return conRol(token, 'admin', function (u) { return accAdminRegistrarRenovacion(u, payload); });
    case 'adminGestionarVendedor': return conRol(token, 'admin', function (u) { return accAdminGestionarVendedor(u, payload); });
    case 'adminResetPassword': return conRol(token, 'admin', function (u) { return accAdminResetPassword(u, payload); });
    case 'adminConfigSet': return conRol(token, 'admin', function (u) { return accAdminConfigSet(u, payload); });
    case 'adminGuardarModulo': return conRol(token, 'admin', function (u) { return accAdminGuardarModulo(u, payload); });

    case 'adminResolverPregunta': return conRol(token, 'admin', function (u) { return accAdminResolverPregunta(u, payload); });

  }
}

/* ============================================================================================
 * 4. HELPERS DE RESPUESTA, SESIÓN Y ROLES
 * ==========================================================================================*/

function respuesta(ok, data, code, error) {
  var payload = ok ? { ok: true, data: data } : { ok: false, error: error || 'Error', code: code || 'ERROR' };
  return ContentService.createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}

/** Ejecuta fn(usuario) solo si hay sesión válida. */
function conSesion(token, fn) {
  var u = verificarSesion(token);
  if (!u) return respuesta(false, null, 'NO_AUTORIZADO', 'Sesión inválida o expirada.');
  return fn(u);
}

/** Ejecuta fn(usuario) solo si la sesión es válida y el rol coincide (admin siempre pasa). */
function conRol(token, rol, fn) {
  var u = verificarSesion(token);
  if (!u) return respuesta(false, null, 'NO_AUTORIZADO', 'Sesión inválida o expirada.');
  if (u.Rol !== rol && u.Rol !== 'admin') {
    return respuesta(false, null, 'PROHIBIDO', 'No tienes permiso para esta acción.');
  }
  return fn(u);
}

/** Verifica un token de sesión; devuelve el objeto usuario o null. */
function verificarSesion(token) {
  if (!token) return null;
  var hash = hashToken(token);
  var sesiones = leerTabla('Sesiones');
  var ahora = Date.now();
  for (var i = 0; i < sesiones.length; i++) {
    var s = sesiones[i];
    if (s.TokenHash === hash && s.Estado === 'activa') {
      if (new Date(s.FechaExpiracion).getTime() < ahora) {
        actualizarFila('Sesiones', 'SessionId', s.SessionId, { Estado: 'expirada' });
        return null;
      }
      var u = buscarFila('Usuarios', 'UserId', s.UserId);
      if (u && u.Estado === 'activo') {
        u.__sessionId = s.SessionId;
        return u;
      }
      return null;
    }
  }
  return null;
}

/* ============================================================================================
 * 5. SEGURIDAD: HASH, TOKENS, SANITIZACIÓN
 * ==========================================================================================*/

function bytesAHex(bytes) {
  var s = '';
  for (var i = 0; i < bytes.length; i++) {
    var b = (bytes[i] + 256) % 256;
    s += (b < 16 ? '0' : '') + b.toString(16);
  }
  return s;
}

function sha256(texto) {
  return bytesAHex(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, texto, Utilities.Charset.UTF_8));
}

function getPepper() {
  return PropertiesService.getScriptProperties().getProperty('PEPPER') || 'arca-pepper-fallback';
}

/** Hash de contraseña: SHA-256 iterado sobre (pepper + salt + password). */
function hashPassword(password, salt) {
  var pepper = getPepper();
  var h = sha256(pepper + salt + password);
  for (var i = 0; i < HASH_ITERACIONES; i++) {
    h = sha256(h + salt);
  }
  return h;
}

function hashToken(token) {
  return sha256(getPepper() + token);
}

function generarSalt() {
  return generarToken(16);
}

/** Token aleatorio hex de n bytes. */
function generarToken(n) {
  var bytes = [];
  for (var i = 0; i < n; i++) bytes.push(Math.floor(Math.random() * 256));
  // Mezcla con un UUID para más entropía.
  return bytesAHex(bytes) + Utilities.getUuid().replace(/-/g, '');
}

/** Contraseña temporal legible pero fuerte. */
function generarPasswordTemporal() {
  var mayus = 'ABCDEFGHJKLMNPQRSTUVWXYZ';
  var minus = 'abcdefghijkmnpqrstuvwxyz';
  var nums = '23456789';
  var simb = '!@#$%&*';
  function pick(s) { return s.charAt(Math.floor(Math.random() * s.length)); }
  var base = pick(mayus) + pick(minus) + pick(nums) + pick(simb);
  for (var i = 0; i < 6; i++) base += pick(mayus + minus + nums);
  return base.split('').sort(function () { return Math.random() - 0.5; }).join('');
}

/** Evita inyección de fórmulas y limpia strings. */
function sanitizar(valor) {
  if (valor === null || valor === undefined) return '';
  var s = String(valor);
  if (/^[=+\-@\t\r]/.test(s)) s = "'" + s; // neutraliza fórmulas
  return s.replace(/[\x00-\x08\x0B\x0C\x0E-\x1F\x7F]/g, '').trim();
}

function idUnico(prefijo) {
  return prefijo + '_' + Date.now().toString(36) + '_' + Math.random().toString(36).slice(2, 8);
}

/* ============================================================================================
 * 6. CAPA DE DATOS GENÉRICA (por nombre de columna)
 * ==========================================================================================*/

function getSheet(nombre) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = ss.getSheetByName(nombre);
  if (!hoja) { asegurarHojas(); hoja = ss.getSheetByName(nombre); }
  return hoja;
}

/** Lee toda la tabla como array de objetos {columna: valor}. */
function leerTabla(nombre) {
  var hoja = getSheet(nombre);
  var ultimaFila = hoja.getLastRow();
  if (ultimaFila < 2) return [];
  var headers = SHEETS[nombre];
  var valores = hoja.getRange(2, 1, ultimaFila - 1, headers.length).getValues();
  return valores.map(function (fila, idx) {
    var obj = { __row: idx + 2 };
    headers.forEach(function (h, i) { obj[h] = fila[i]; });
    return obj;
  });
}

/** Devuelve la primera fila cuyo columna == valor, o null. */
function buscarFila(nombre, columna, valor) {
  var filas = leerTabla(nombre);
  for (var i = 0; i < filas.length; i++) {
    if (String(filas[i][columna]) === String(valor)) return filas[i];
  }
  return null;
}

function filtrarFilas(nombre, columna, valor) {
  return leerTabla(nombre).filter(function (f) { return String(f[columna]) === String(valor); });
}

/** Inserta una fila a partir de un objeto {columna: valor}. */
function insertarFila(nombre, obj) {
  var hoja = getSheet(nombre);
  var headers = SHEETS[nombre];
  var fila = headers.map(function (h) {
    return obj[h] === undefined ? '' : (typeof obj[h] === 'string' ? sanitizar(obj[h]) : obj[h]);
  });
  hoja.appendRow(fila);
  return obj;
}

/** Actualiza la fila identificada por columna==valor con los campos de cambios. */
function actualizarFila(nombre, columna, valor, cambios) {
  var hoja = getSheet(nombre);
  var fila = buscarFila(nombre, columna, valor);
  if (!fila) return false;
  var headers = SHEETS[nombre];
  headers.forEach(function (h, i) {
    if (cambios.hasOwnProperty(h)) {
      var v = typeof cambios[h] === 'string' ? sanitizar(cambios[h]) : cambios[h];
      hoja.getRange(fila.__row, i + 1).setValue(v);
    }
  });
  return true;
}

/* ============================================================================================
 * 7. CONFIGURACIÓN (hoja Configuracion)
 * ==========================================================================================*/

function configGet(clave) {
  var f = buscarFila('Configuracion', 'Clave', clave);
  return f ? String(f.Valor) : null;
}
function configNum(clave, defecto) {
  var v = configGet(clave);
  return v === null ? defecto : parseFloat(v);
}
function configSet(clave, valor, descripcion) {
  var existe = buscarFila('Configuracion', 'Clave', clave);
  if (existe) {
    actualizarFila('Configuracion', 'Clave', clave, { Valor: valor, FechaActualizacion: new Date() });
  } else {
    insertarFila('Configuracion', {
      Clave: clave, Valor: valor, Descripcion: descripcion || '', FechaActualizacion: new Date()
    });
  }
}
function configPublica() {
  return {
    comision_primera_venta: configNum('comision_primera_venta', 0.15),
    comision_renovacion: configNum('comision_renovacion', 0.05),
    puntaje_minimo: configNum('puntaje_minimo', 85),
    intentos_por_examen: configNum('intentos_por_examen', 3),
    intentos_examen_final: configNum('intentos_examen_final', 2),
    version_terminos: configGet('version_terminos') || '2026.08',
    moneda: configGet('moneda') || 'USD',
    whatsapp_soporte: configGet('whatsapp_soporte') || ''
  };
}

function auditar(userId, accion, entidad, entityId, antes, despues, ip) {
  insertarFila('Auditoria', {
    AuditId: idUnico('aud'), UserId: userId || '', Accion: accion, Entidad: entidad || '',

  });
}

function crearUsuario(email, password, rol, candidateId, debeCambiar) {
  email = String(email).toLowerCase().trim();
  if (buscarFila('Usuarios', 'Email', email)) {
    throw new Error('El correo ya está registrado.');
  }
  var salt = generarSalt();
  var user = {
    UserId: idUnico('usr'), CandidateId: candidateId || '', Email: email,
    PasswordHash: hashPassword(password, salt), Salt: salt, Rol: rol || 'candidato',
    Estado: 'activo', DebeCambiarPassword: debeCambiar ? 'true' : 'false',
    UltimoAcceso: '', IntentosFallidos: 0, BloqueadoHasta: '', FechaCreacion: new Date()
  };
  insertarFila('Usuarios', user);
  return user;
}

function accLogin(payload, ctx) {
  var email = String(payload.email || '').toLowerCase().trim();
  var password = String(payload.password || '');
  if (!email || !password) return respuesta(false, null, 'DATOS', 'Correo y contraseña son obligatorios.');

  var u = buscarFila('Usuarios', 'Email', email);
  if (!u) return respuesta(false, null, 'CREDENCIALES', 'Correo o contraseña incorrectos.');

  if (u.BloqueadoHasta && new Date(u.BloqueadoHasta).getTime() > Date.now()) {
    return respuesta(false, null, 'BLOQUEADO', 'Cuenta bloqueada temporalmente. Intenta más tarde.');
  }
  if (u.Estado !== 'activo') {
    return respuesta(false, null, 'INACTIVO', 'Tu cuenta no está activa. Contacta a soporte.');
  }

  var hash = hashPassword(password, u.Salt);
  if (hash !== u.PasswordHash) {
    var intentos = Number(u.IntentosFallidos || 0) + 1;
    var cambios = { IntentosFallidos: intentos };
    if (intentos >= MAX_INTENTOS_LOGIN) {
      cambios.BloqueadoHasta = new Date(Date.now() + BLOQUEO_MS);
      cambios.IntentosFallidos = 0;
    }
    actualizarFila('Usuarios', 'UserId', u.UserId, cambios);
    auditar(u.UserId, 'login_fallido', 'usuario', u.UserId, null, null, ctx.ip);
    return respuesta(false, null, 'CREDENCIALES', 'Correo o contraseña incorrectos.');
  }

  actualizarFila('Usuarios', 'UserId', u.UserId, { IntentosFallidos: 0, BloqueadoHasta: '', UltimoAcceso: new Date() });
  var token = crearSesion(u.UserId, ctx);
  auditar(u.UserId, 'login', 'usuario', u.UserId, null, null, ctx.ip);

  return respuesta(true, {
    token: token,
    usuario: usuarioPublico(u),
    debeCambiarPassword: String(u.DebeCambiarPassword) === 'true'
  });
}

function crearSesion(userId, ctx) {
  var token = generarToken(24);
  insertarFila('Sesiones', {
    SessionId: idUnico('ses'), UserId: userId, TokenHash: hashToken(token),
    FechaCreacion: new Date(), FechaExpiracion: new Date(Date.now() + SESION_DURACION_MS),
    Estado: 'activa', IP: (ctx && ctx.ip) || '', UserAgent: (ctx && ctx.userAgent) || ''
  });
  return token;
}

function accCambiarPassword(payload, token) {
  var u = verificarSesion(token);
  if (!u) return respuesta(false, null, 'NO_AUTORIZADO', 'Sesión inválida.');
  var actual = String(payload.actual || '');
  var nueva = String(payload.nueva || '');
  if (nueva.length < 8) return respuesta(false, null, 'DEBIL', 'La nueva contraseña debe tener al menos 8 caracteres.');

  // Si NO debe cambiar obligatoriamente, exige la contraseña actual.
  if (String(u.DebeCambiarPassword) !== 'true') {
    if (hashPassword(actual, u.Salt) !== u.PasswordHash) {
      return respuesta(false, null, 'CREDENCIALES', 'La contraseña actual es incorrecta.');
    }
  }
  var salt = generarSalt();
  actualizarFila('Usuarios', 'UserId', u.UserId, {
    PasswordHash: hashPassword(nueva, salt), Salt: salt, DebeCambiarPassword: 'false'
  });
  auditar(u.UserId, 'cambio_password', 'usuario', u.UserId, null, null, '');
  return respuesta(true, { actualizada: true });
}

function accRegistrarCandidato(payload, ctx) {
  var email = String(payload.email || '').toLowerCase().trim();
  var whatsapp = String(payload.whatsapp || '').replace(/[^\d+]/g, '');

  if (!payload.nombreCompleto || !email || !whatsapp) {
    return respuesta(false, null, 'DATOS', 'Nombre, correo y WhatsApp son obligatorios.');
  }
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
    return respuesta(false, null, 'EMAIL', 'El formato del correo no es válido.');
  }
  if (buscarFila('Candidatos', 'Email', email) || buscarFila('Usuarios', 'Email', email)) {
    return respuesta(false, null, 'DUPLICADO', 'Este correo ya está registrado.');
  }
  if (buscarFila('Candidatos', 'WhatsApp', whatsapp)) {
    return respuesta(false, null, 'DUPLICADO', 'Este WhatsApp ya está registrado.');
  }
  if (!payload.consentimientoDatos || !payload.confirmaVeracidad) {
    return respuesta(false, null, 'CONSENTIMIENTO', 'Debes aceptar los consentimientos iniciales.');
  }
  // Edad mínima
  if (payload.fechaNacimiento) {
    var edad = edadDesde(payload.fechaNacimiento);
    if (edad < 18) return respuesta(false, null, 'EDAD', 'Debes ser mayor de edad para participar.');
  }

  var candidateId = idUnico('cand');
  insertarFila('Candidatos', {
    CandidateId: candidateId, NombreCompleto: payload.nombreCompleto, Pais: payload.pais || '',
    Ciudad: payload.ciudad || '', FechaNacimiento: payload.fechaNacimiento || '', Email: email,
    WhatsApp: whatsapp, Documento: payload.documento || '', ZonaHoraria: payload.zonaHoraria || '',
    Idioma: payload.idioma || 'es', FuenteConocio: payload.fuenteConocio || '',
    Experiencia: payload.experiencia || '', ExperienciaSoftware: payload.experienciaSoftware || '',
    Sectores: payload.sectores || '', Disponibilidad: payload.disponibilidad || '',
    MedioContacto: payload.medioContacto || '', Motivacion: payload.motivacion || '',
    Estado: 'en_capacitacion', FechaRegistro: new Date(), UltimaActividad: new Date(),
    Progreso: 0, Certificado: 'false'
  });

  // Crea el usuario candidato (correo personal como acceso, contraseña que eligió).
  if (!payload.password || String(payload.password).length < 8) {
    return respuesta(false, null, 'DEBIL', 'La contraseña debe tener al menos 8 caracteres.');
  }
  crearUsuario(email, payload.password, 'candidato', candidateId, false);

  var token = crearSesion(buscarFila('Usuarios', 'Email', email).UserId, ctx);
  auditar('', 'registro_candidato', 'candidato', candidateId, null, { email: email }, ctx.ip);

  return respuesta(true, { candidateId: candidateId, token: token, email: email });
}

function edadDesde(fecha) {
  var nac = new Date(fecha);
  if (isNaN(nac.getTime())) return 99;
  var hoy = new Date();
  var edad = hoy.getFullYear() - nac.getFullYear();
  var m = hoy.getMonth() - nac.getMonth();
  if (m < 0 || (m === 0 && hoy.getDate() < nac.getDate())) edad--;
  return edad;
}

function accGuardarProgreso(u, payload) {
  var moduleId = String(payload.moduleId || '');
  if (!moduleId) return respuesta(false, null, 'DATOS', 'moduleId es obligatorio.');
  var porcentaje = Math.max(0, Math.min(100, Number(payload.porcentaje || 0)));
  var estado = payload.estado || (porcentaje >= 100 ? 'completado' : 'en_progreso');

  var existente = null;
  var progreso = filtrarFilas('Progreso', 'CandidateId', u.CandidateId);
  for (var i = 0; i < progreso.length; i++) {
    if (String(progreso[i].ModuleId) === moduleId) { existente = progreso[i]; break; }
  }
  if (existente) {
    actualizarFila('Progreso', 'ProgressId', existente.ProgressId, {
      Estado: estado, Porcentaje: porcentaje, UltimaActividad: new Date(),
      FechaFinalizacion: estado === 'completado' ? new Date() : existente.FechaFinalizacion
    });
  } else {
    insertarFila('Progreso', {
      ProgressId: idUnico('prg'), CandidateId: u.CandidateId, ModuleId: moduleId, Estado: estado,
      Porcentaje: porcentaje, FechaInicio: new Date(),
      FechaFinalizacion: estado === 'completado' ? new Date() : '', UltimaActividad: new Date()
    });
  }
  actualizarProgresoGlobal(u.CandidateId);
  return respuesta(true, { guardado: true });
}

function actualizarProgresoGlobal(candidateId) {
  var modulos = leerTabla('Modulos').filter(function (m) { return String(m.Obligatorio) === 'true'; });
  var progreso = filtrarFilas('Progreso', 'CandidateId', candidateId);
  var completados = 0;
  modulos.forEach(function (m) {
    for (var i = 0; i < progreso.length; i++) {
      if (String(progreso[i].ModuleId) === String(m.ModuleId) && String(progreso[i].Estado) === 'completado') {
        completados++; break;
      }
    }
  });
  var pct = modulos.length ? Math.round((completados / modulos.length) * 100) : 0;
  actualizarFila('Candidatos', 'CandidateId', candidateId, { Progreso: pct, UltimaActividad: new Date() });
}

/** Devuelve las preguntas de un módulo SIN la respuesta correcta (anti-trampa). */
function accObtenerExamen(u, payload) {
  var moduleId = String(payload.moduleId || '');
  var preguntas = filtrarFilas('Preguntas', 'ModuleId', moduleId)
    .filter(function (p) { return String(p.Estado) === 'activo' || p.Estado === ''; })
    .map(function (p) {
      return {
        QuestionId: p.QuestionId, Tipo: p.Tipo, Pregunta: p.Pregunta,
        Opciones: p.Opciones, Puntaje: p.Puntaje
      };
    });
  // Mezcla y limita si se solicita un tamaño.
  preguntas = mezclar(preguntas);
  if (payload.cantidad) preguntas = preguntas.slice(0, Number(payload.cantidad));
  var intentosPrevios = filtrarFilas('IntentosExamen', 'CandidateId', u.CandidateId)
    .filter(function (a) { return String(a.ModuleId) === moduleId; }).length;
  var maxIntentos = configNum('intentos_por_examen', 3);
  if (String(moduleId) === 'final') maxIntentos = configNum('intentos_examen_final', 2);
  if (intentosPrevios >= maxIntentos) {
    return respuesta(false, null, 'SIN_INTENTOS', 'Alcanzaste el máximo de intentos para este examen.');
  }
  return respuesta(true, { preguntas: preguntas, intentosRestantes: maxIntentos - intentosPrevios });
}

/** Califica el examen en el backend (nunca en el frontend). */
function accEnviarExamen(u, payload) {
  var moduleId = String(payload.moduleId || '');
  var respuestas = payload.respuestas || {}; // { QuestionId: valor }
  var duracion = Number(payload.duracion || 0);

  var maxIntentos = String(moduleId) === 'final' ? configNum('intentos_examen_final', 2) : configNum('intentos_por_examen', 3);
  var intentosPrevios = filtrarFilas('IntentosExamen', 'CandidateId', u.CandidateId)
    .filter(function (a) { return String(a.ModuleId) === moduleId; }).length;
  if (intentosPrevios >= maxIntentos) {
    return respuesta(false, null, 'SIN_INTENTOS', 'Sin intentos disponibles.');
  }

  var banco = filtrarFilas('Preguntas', 'ModuleId', moduleId);
  var totalPuntos = 0, obtenidos = 0, detalle = [];
  banco.forEach(function (p) {
    if (respuestas[p.QuestionId] === undefined) return; // no incluida en este intento
    var puntos = Number(p.Puntaje || 1);
    totalPuntos += puntos;
    var correcta = calificarRespuesta(p, respuestas[p.QuestionId]);
    if (correcta) obtenidos += puntos;
    detalle.push({ QuestionId: p.QuestionId, correcta: correcta, explicacion: p.Explicacion });
  });

  var puntaje = totalPuntos ? Math.round((obtenidos / totalPuntos) * 100) : 0;
  var minimo = String(moduleId) === 'final'
    ? configNum('puntaje_minimo', 85)
    : Number((buscarFila('Modulos', 'ModuleId', moduleId) || {}).PuntajeMinimo || configNum('puntaje_minimo', 85));
  var aprobado = puntaje >= minimo;

  insertarFila('IntentosExamen', {
    AttemptId: idUnico('att'), CandidateId: u.CandidateId, ModuleId: moduleId, Puntaje: puntaje,
    Aprobado: aprobado ? 'true' : 'false', Respuestas: JSON.stringify(respuestas),
    FechaInicio: payload.fechaInicio || '', FechaFinalizacion: new Date(), Duracion: duracion
  });

  if (aprobado && moduleId !== 'final') {
    accGuardarProgreso(u, { moduleId: moduleId, porcentaje: 100, estado: 'completado' });
  }

  return respuesta(true, {
    puntaje: puntaje, aprobado: aprobado, minimo: minimo, detalle: detalle,
    intentosRestantes: maxIntentos - (intentosPrevios + 1)
  });
}

function calificarRespuesta(pregunta, respuesta) {
  var tipo = String(pregunta.Tipo || 'opcion');
  var correcta = String(pregunta.RespuestaCorrecta || '').trim();
  if (tipo === 'abierta') {
    // Coincidencia por palabras clave (correcta = lista separada por ;)
    var claves = correcta.toLowerCase().split(';').map(function (s) { return s.trim(); }).filter(Boolean);
    var texto = String(respuesta || '').toLowerCase();
    var aciertos = claves.filter(function (c) { return texto.indexOf(c) >= 0; }).length;
    return claves.length > 0 && aciertos >= Math.ceil(claves.length / 2);
  }
  return String(respuesta).trim().toLowerCase() === correcta.toLowerCase();
}

function mezclar(arr) {
  var a = arr.slice();
  for (var i = a.length - 1; i > 0; i--) {
    var j = Math.floor(Math.random() * (i + 1));
    var t = a[i]; a[i] = a[j]; a[j] = t;
  }
  return a;
}

/* ============================================================================================
 * 12. CERTIFICACIÓN Y CREACIÓN DEL VENDEDOR
 * ==========================================================================================*/

function evaluarRequisitos(candidateId) {
  var modulos = leerTabla('Modulos').filter(function (m) { return String(m.Obligatorio) === 'true'; });
  var progreso = filtrarFilas('Progreso', 'CandidateId', candidateId);
  var intentos = filtrarFilas('IntentosExamen', 'CandidateId', candidateId);
  var terminos = filtrarFilas('TerminosAceptados', 'CandidateId', candidateId);
  var simulaciones = filtrarFilas('Simulaciones', 'CandidateId', candidateId);

  function moduloCompleto(id) {
    return progreso.some(function (p) { return String(p.ModuleId) === String(id) && String(p.Estado) === 'completado'; });
  }
  function examenAprobado(id) {
    return intentos.some(function (a) { return String(a.ModuleId) === String(id) && String(a.Aprobado) === 'true'; });
  }

  var modulosCompletos = modulos.length > 0 && modulos.every(function (m) { return moduloCompleto(m.ModuleId); });
  var examenesAprobados = modulos.length > 0 && modulos.every(function (m) { return examenAprobado(m.ModuleId); });
  var examenFinal = examenAprobado('final');
  var docsRequeridos = ['terminos', 'comisiones', 'privacidad', 'conducta'];
  var terminosOk = docsRequeridos.every(function (d) {
    return terminos.some(function (t) { return String(t.TipoDocumento) === d; });
  });
  var simulacionesOk = simulaciones.length >= 3;

  var mejorFinal = 0;
  intentos.forEach(function (a) {
    if (String(a.ModuleId) === 'final') mejorFinal = Math.max(mejorFinal, Number(a.Puntaje || 0));
  });
  var puntajeMinimo = configNum('puntaje_minimo', 85);

  var requisitos = {
    modulosCompletos: modulosCompletos,
    examenesAprobados: examenesAprobados,
    examenFinalAprobado: examenFinal,
    terminosAceptados: terminosOk,
    simulacionesCompletas: simulacionesOk,
    puntajeFinal: mejorFinal,
    puntajeMinimo: puntajeMinimo,
    puntajeSuficiente: mejorFinal >= puntajeMinimo
  };
  requisitos.cumpleTodo = modulosCompletos && examenesAprobados && examenFinal &&
    terminosOk && simulacionesOk && requisitos.puntajeSuficiente;
  return requisitos;
}

function accEstadoCertificacion(u) {
  return respuesta(true, evaluarRequisitos(u.CandidateId));
}

function accCertificar(u, ctx) {
  var candidateId = u.CandidateId;
  var req = evaluarRequisitos(candidateId);
  if (!req.cumpleTodo) {
    return respuesta(false, req, 'REQUISITOS', 'Aún no cumples todos los requisitos de certificación.');
  }
  // Idempotencia: si ya es vendedor, no recrear.
  if (buscarFila('Vendedores', 'CandidateId', candidateId)) {
    return respuesta(false, null, 'YA_CERTIFICADO', 'Ya estás certificado.');
  }

  var cand = buscarFila('Candidatos', 'CandidateId', candidateId);
  var codigoVendedor = generarCodigoVendedor();
  var codigoReferido = codigoVendedor + '-REF';
  var codigoCert = 'ARCA-' + new Date().getFullYear() + '-' + codigoVendedor;

  insertarFila('Certificados', {
    CertificateId: idUnico('cert'), CandidateId: candidateId, CodigoCertificado: codigoCert,
    FechaEmision: new Date(), PuntajeFinal: req.puntajeFinal, Estado: 'vigente',
    UrlVerificacion: '?action=verificarCertificado&codigo=' + encodeURIComponent(codigoCert)
  });

  insertarFila('Vendedores', {
    SellerId: idUnico('sell'), CandidateId: candidateId, CodigoVendedor: codigoVendedor,
    CodigoReferido: codigoReferido, Nivel: 'inicial', Estado: 'activo',
    FechaCertificacion: new Date(), ClientesActivos: 0, VentasTotales: 0
  });

  // Contraseña temporal + cambio de rol a vendedor. El correo personal es el usuario.
  var passwordTemporal = generarPasswordTemporal();
  var salt = generarSalt();
  actualizarFila('Usuarios', 'UserId', u.UserId, {
    Rol: 'vendedor', PasswordHash: hashPassword(passwordTemporal, salt), Salt: salt,
    DebeCambiarPassword: 'true'
  });
  actualizarFila('Candidatos', 'CandidateId', candidateId, { Estado: 'certificado', Certificado: 'true' });

  enviarEmailCredenciales(u.Email, passwordTemporal, codigoVendedor);
  auditar(u.UserId, 'certificacion', 'vendedor', codigoVendedor, null,
    { codigoCertificado: codigoCert, puntaje: req.puntajeFinal }, ctx.ip);

  // La contraseña temporal se devuelve UNA sola vez (no se vuelve a mostrar).
  return respuesta(true, {
    codigoVendedor: codigoVendedor, codigoReferido: codigoReferido,
    codigoCertificado: codigoCert, email: u.Email, passwordTemporal: passwordTemporal
  });
}

function generarCodigoVendedor() {
  var n = leerTabla('Vendedores').length + 1;
  var base = 'V' + ('0000' + n).slice(-4);
  // Asegura unicidad.
  while (buscarFila('Vendedores', 'CodigoVendedor', base)) {
    n++; base = 'V' + ('0000' + n).slice(-4);
  }
  return base;
}

function accDashboardVendedor(u) {
  var v = sellerDeUsuario(u);
  if (!v) return respuesta(false, null, 'NO_VENDEDOR', 'No se encontró tu perfil de vendedor.');
  var prospectos = filtrarFilas('Prospectos', 'SellerId', v.SellerId);
  var ventas = filtrarFilas('Ventas', 'SellerId', v.SellerId);
  var comisiones = filtrarFilas('Comisiones', 'SellerId', v.SellerId);

  var ganados = prospectos.filter(function (p) { return String(p.Etapa) === 'ganado'; }).length;
  var ventasAprobadas = ventas.filter(function (s) { return String(s.Estado) === 'aprobada'; });
  var comisionesPendientes = comisiones.filter(function (c) { return String(c.Estado) === 'pendiente' || String(c.Estado) === 'aprobada'; });
  var comisionesPagadas = comisiones.filter(function (c) { return String(c.Estado) === 'pagada'; });

  return respuesta(true, {
    vendedor: {
      codigoVendedor: v.CodigoVendedor, codigoReferido: v.CodigoReferido, nivel: v.Nivel,
      estado: v.Estado, fechaCertificacion: v.FechaCertificacion
    },
    kpis: {
      prospectos: prospectos.length,
      clientesActivos: ganados,
      ventasCerradas: ventasAprobadas.length,
      comisionesPendientes: suma(comisionesPendientes),
      comisionesPagadas: suma(comisionesPagadas),
      tasaConversion: prospectos.length ? Math.round((ganados / prospectos.length) * 100) : 0
    }
  });
}

function accCrearProspecto(u, payload) {
  if (String(configGet('mantenimiento')) === '1') {
    return respuesta(false, null, 'MANTENIMIENTO', 'Sistema en mantenimiento. Intenta más tarde.');
  }
  var v = sellerDeUsuario(u);
  if (!payload.empresa && !payload.contacto) {
    return respuesta(false, null, 'DATOS', 'Indica al menos empresa o contacto.');
  }
  // Anti-duplicado: mismo email o whatsapp en cualquier vendedor => revisión.
  var email = String(payload.email || '').toLowerCase().trim();
  var whatsapp = String(payload.whatsapp || '').replace(/[^\d+]/g, '');
  var dup = leerTabla('Prospectos').filter(function (p) {
    return (email && String(p.Email).toLowerCase() === email) ||
      (whatsapp && String(p.WhatsApp).replace(/[^\d+]/g, '') === whatsapp);
  });
  if (dup.length) {
    return respuesta(false, { duplicados: dup.length }, 'DUPLICADO',
      'Ya existe un prospecto con ese contacto. Un administrador revisará la atribución.');
  }

  var prospectId = idUnico('pros');
  insertarFila('Prospectos', {
    ProspectId: prospectId, SellerId: v.SellerId, Empresa: payload.empresa || '',
    Contacto: payload.contacto || '', Email: email, WhatsApp: whatsapp, Pais: payload.pais || '',
    Sector: payload.sector || '', Fuente: payload.fuente || '', Etapa: payload.etapa || 'nuevo',
    ValorEstimado: Number(payload.valorEstimado || 0), ProximaAccion: payload.proximaAccion || '',
    FechaSeguimiento: payload.fechaSeguimiento || '', Notas: payload.notas || '',
    FechaCreacion: new Date(), FechaActualizacion: new Date()
  });
  return respuesta(true, { prospectId: prospectId });
}

/** Calcula (backend) la comisión de una venta aprobada. Fuente del %: hoja Configuracion. */
function calcularComisionDeVenta(venta) {
  var esRenovacion = String(venta.TipoVenta) === 'renovacion';
  var pct = esRenovacion ? configNum('comision_renovacion', 0.05) : configNum('comision_primera_venta', 0.15);
  var monto = +(Number(venta.Monto || 0) * pct).toFixed(2);
  var commissionId = idUnico('com');
  insertarFila('Comisiones', {
    CommissionId: commissionId, SaleId: venta.SaleId, SellerId: venta.SellerId,
    Tipo: esRenovacion ? 'renovacion' : 'primera', Porcentaje: pct, Monto: monto,
    Estado: 'pendiente', FechaCreacion: new Date(),
    FechaProgramada: new Date(Date.now() + 1000 * 60 * 60 * 24 * 30), FechaPago: '', MetodoPago: ''
  });
  return { commissionId: commissionId, monto: monto, porcentaje: pct };
}

var HOJAS_ADMIN_LEGIBLES = ['Candidatos', 'Usuarios', 'Vendedores', 'Ventas', 'Comisiones',
  'Prospectos', 'Renovaciones', 'IntentosExamen', 'Simulaciones', 'TerminosAceptados',
  'Certificados', 'ChatbotLogs', 'PreguntasNoResueltas', 'Auditoria', 'Modulos', 'Preguntas',
  'Configuracion', 'Progreso'];

function accAdminListar(payload) {
  var hoja = String(payload.hoja || '');
  if (HOJAS_ADMIN_LEGIBLES.indexOf(hoja) < 0) {
    return respuesta(false, null, 'HOJA_INVALIDA', 'Hoja no permitida.');
  }
  var filas = leerTabla(hoja);
  // No exponer hashes/salt.
  if (hoja === 'Usuarios') {
    filas = filas.map(function (f) {
      var c = {}; for (var k in f) if (k !== 'PasswordHash' && k !== 'Salt') c[k] = f[k];
      return c;
    });
  }
  if (payload.filtroColumna && payload.filtroValor !== undefined) {
    filas = filas.filter(function (f) {
      return String(f[payload.filtroColumna]) === String(payload.filtroValor);
    });
  }
  return respuesta(true, filas);
}

function accAdminActualizarVenta(u, payload) {
  var saleId = String(payload.saleId || '');
  var estado = String(payload.estado || '');
  if (['pendiente', 'en_revision', 'aprobada', 'rechazada', 'cancelada', 'reembolsada'].indexOf(estado) < 0) {
    return respuesta(false, null, 'DATOS', 'Estado inválido.');
  }
  var venta = buscarFila('Ventas', 'SaleId', saleId);
  if (!venta) return respuesta(false, null, 'NO_ENCONTRADO', 'Venta no encontrada.');
  var previo = { estado: venta.Estado };
  actualizarFila('Ventas', 'SaleId', saleId, {
    Estado: estado, FechaValidacion: new Date(), ValidadoPor: u.Email,
    Comprobante: payload.observacion ? (venta.Comprobante + ' | ' + payload.observacion) : venta.Comprobante
  });

  var comisionInfo = null;
  // Al aprobar, se genera la comisión (si no existe ya para esa venta).
  if (estado === 'aprobada') {
    var existe = filtrarFilas('Comisiones', 'SaleId', saleId).length > 0;
    if (!existe) {
      venta.Estado = 'aprobada';
      comisionInfo = calcularComisionDeVenta(venta);
    }
    // Actualiza contadores del vendedor.
    var vend = buscarFila('Vendedores', 'SellerId', venta.SellerId);
    if (vend) {
      var ventasAprobadas = filtrarFilas('Ventas', 'SellerId', venta.SellerId)
        .filter(function (s) { return String(s.Estado) === 'aprobada'; }).length;
      actualizarFila('Vendedores', 'SellerId', vend.SellerId, {
        VentasTotales: ventasAprobadas, ClientesActivos: ventasAprobadas
      });
    }
  }
  // Al reembolsar/cancelar, anula comisiones asociadas.
  if (estado === 'reembolsada' || estado === 'cancelada') {
    filtrarFilas('Comisiones', 'SaleId', saleId).forEach(function (c) {
      actualizarFila('Comisiones', 'CommissionId', c.CommissionId, { Estado: 'anulada' });
    });
  }
  auditar(u.UserId, 'admin_venta_' + estado, 'venta', saleId, previo, { estado: estado }, '');
  return respuesta(true, { estado: estado, comision: comisionInfo });
}

    FechaEmision: new Date(), PuntajeFinal: 92, Estado: 'vigente', UrlVerificacion: ''

    Nivel: 'inicial', Estado: 'activo', FechaCertificacion: new Date(), ClientesActivos: 2, VentasTotales: 2

  var prospectos = [
